# DSL v4.5.9 — 独立 crontab (替代 OpenClaw cron)
# 部署: crontab deploy/crontab.txt
# 验证: crontab -l

DSL_ROOT=/opt/dsl-quant
PYTHON=/usr/bin/python3

# ═══════════ 凌晨 ═══════════
# 02:10 DSL系统备份

20 9 * * 1-5 cd $DSL_ROOT && $PYTHON scripts/morning_decision.py --market a --mode morning >> $DSL_ROOT/logs/morning_decision.log 2>&1

# 17:00 个股预测
0 17 * * 1-5 cd $DSL_ROOT && $PYTHON scripts/batch_predict.py >> $DSL_ROOT/logs/predict.log 2>&1

# 20:00 自我反思

0 21 * * 0-4 cd $DSL_ROOT && $PYTHON scripts/blackswan_review.py >> $DSL_ROOT/logs/blackswan.log 2>&1
# 21:30 盘前交易预案
30 21 * * 0-4 cd $DSL_ROOT && $PYTHON scripts/morning_decision.py --market a --mode evening >> $DSL_ROOT/logs/plan.log 2>&1

# 周六 09:00 周度复盘
0 9 * * 6 cd $DSL_ROOT && $PYTHON scripts/weekly_review.py >> $DSL_ROOT/logs/weekly.log 2>&1
